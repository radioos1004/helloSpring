version: '3.8'

services:
  app:
    build:
      context: ./demo  # Dockerfile이 있는 폴더 (build/libs 기준)
      dockerfile: Dockerfile
    image: hello-spring:latest
    container_name: spring-app
    ports:
      - "8081:8081"  # 호스트:컨테이너 (Dockerfile EXPOSE 8081)
    environment:
      - SPRING_PROFILES_ACTIVE=prod
      # 추가 환경변수 필요시 여기에
      - SERVER_PORT=8081
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "wget", "--no-verbose", "--tries=1", "--spider", "http://localhost:8081/actuator/health"]
      interval: 30s
      timeout: 3s
      start_period: 60s
      retries: 3
